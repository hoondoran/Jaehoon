        var out = {hq: [], br: []};
        var s = props && props.reg_26h; if(!s) return out;
        var re = /([^\/()]+)\((본사|지사)(\d+)\)/g, m;
        while((m = re.exec(s)) !== null){
            (m[2] === '본사' ? out.hq : out.br).push({name: m[1].trim(), cnt: Number(m[3])});
        }
        return out;
    }
